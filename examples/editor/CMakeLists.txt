set(FFMPEG_ROOT "${PROJECT_SOURCE_DIR}/third_party/ffmpeg" CACHE PATH "FFmpeg shared development distribution")
if(NOT EXISTS "${FFMPEG_ROOT}/include/libavformat/avformat.h")
  message(FATAL_ERROR "Silta Cut requires FFmpeg headers, import libraries and DLLs in FFMPEG_ROOT")
endif()
add_library(cut_media media.cpp)
target_include_directories(cut_media PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE "${FFMPEG_ROOT}/include")
foreach(lib avformat avcodec avutil swscale swresample)
  find_library(CUT_${lib} NAMES ${lib} PATHS "${FFMPEG_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
  target_link_libraries(cut_media PUBLIC "${CUT_${lib}}")
endforeach()
function(cut_runtime target)
  file(GLOB dlls "${FFMPEG_ROOT}/bin/*.dll")
  add_custom_command(TARGET ${target} POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different ${dlls} $<TARGET_FILE_DIR:${target}>)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8 /permissive-)
    target_compile_definitions(${target} PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
  endif()
endfunction()
add_executable(cut_media_test tests.cpp)
target_link_libraries(cut_media_test PRIVATE cut_media)
cut_runtime(cut_media_test)

# miniaudio 0.11.23, header-only. Fetch only when absent from the local cache.
set(MINIAUDIO_DIR "${PROJECT_SOURCE_DIR}/third_party/miniaudio")
if(NOT EXISTS "${MINIAUDIO_DIR}/miniaudio.h")
  file(MAKE_DIRECTORY "${MINIAUDIO_DIR}")
  file(DOWNLOAD https://raw.githubusercontent.com/mackron/miniaudio/0.11.23/miniaudio.h
    "${MINIAUDIO_DIR}/miniaudio.h" STATUS download_status TLS_VERIFY ON
    EXPECTED_HASH SHA256=7e4f3f13c8fe66df2080ac3dd12a89193e3c2463cb7f067c798abd7331cd8ee6)
  list(GET download_status 0 download_code)
  if(NOT download_code EQUAL 0)
    file(REMOVE "${MINIAUDIO_DIR}/miniaudio.h")
    message(FATAL_ERROR "Download miniaudio 0.11.23 failed: ${download_status}")
  endif()
endif()
file(SHA256 "${MINIAUDIO_DIR}/miniaudio.h" miniaudio_hash)
if(NOT miniaudio_hash STREQUAL "7e4f3f13c8fe66df2080ac3dd12a89193e3c2463cb7f067c798abd7331cd8ee6")
  message(FATAL_ERROR "Silta Cut expects the unmodified miniaudio 0.11.23 header")
endif()
add_library(cut_engine engine.cpp export.cpp)
target_link_libraries(cut_engine PUBLIC cut_media)
target_include_directories(cut_engine PRIVATE "${MINIAUDIO_DIR}" "${FFMPEG_ROOT}/include")
if(MSVC)
  target_compile_definitions(cut_engine PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
  target_compile_options(cut_engine PRIVATE /utf-8)
endif()
add_executable(silta_cut WIN32 main.cpp)
target_link_libraries(silta_cut PRIVATE cut_engine Silta::app)
if(MSVC)
  target_link_options(silta_cut PRIVATE /ENTRY:mainCRTStartup)
endif()
cut_runtime(silta_cut)
add_custom_command(TARGET silta_cut POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${PROJECT_SOURCE_DIR}/assets/fonts" $<TARGET_FILE_DIR:silta_cut>/fonts)
